5. GUIDE D'INTÉGRATION RAPIDE
═══════════════════════════════════════════════════════════════════════════════

ÉTAPE 1: Importer le hook
--------------------------
import { useNotifications } from '../../components/NotificationProvider'

ÉTAPE 2: Déclarer dans le composant
------------------------------------
const { success, error, warning, info, confirm, confirmDelete, confirmExit } = useNotifications()

ÉTAPE 3: Remplacer les alert()
-------------------------------
AVANT:  alert('Message')
APRÈS:  error('Message')  OU  warning('Message')  OU  info('Message')

ÉTAPE 4: Remplacer les confirm()
---------------------------------
AVANT:  if (confirm('Supprimer ?')) { ... }
APRÈS:  const confirmed = await confirmDelete(item.nom)
        if (confirmed) { ... }

ÉTAPE 5: Ajouter success() après actions réussies
--------------------------------------------------
try {
  await saveData()
  success('Données enregistrées avec succès')
} catch (err) {
  error('Erreur lors de l\'enregistrement')
}

6. EXEMPLES DE CODE
═══════════════════════════════════════════════════════════════════════════════

A. CRÉATION D'UN ÉLÉMENT
-------------------------
const handleCreate = async () => {
  if (!formData.nom) {
    error('Le nom est obligatoire')
    return
  }
  
  try {
    await createItem(formData)
    success(`"${formData.nom}" créé avec succès`)
    closeModal()
  } catch (err) {
    error('Erreur lors de la création')
  }
}

B. SUPPRESSION AVEC CONFIRMATION
---------------------------------
const handleDelete = async (item) => {
  const confirmed = await confirmDelete(item.nom)
  if (confirmed) {
    try {
      await deleteItem(item.id)
      success(`"${item.nom}" supprimé avec succès`)
    } catch (err) {
      error('Erreur lors de la suppression')
    }
  }
}

C. VALIDATION DE FORMULAIRE
----------------------------
const handleValidate = async () => {
  const confirmed = await confirm({
    title: 'Valider le document',
    message: 'Cette action est irréversible. Continuer ?',
    type: 'warning',
    confirmText: 'Valider',
    cancelText: 'Annuler'
  })
  
  if (confirmed) {
    try {
      await validateDocument()
      success('Document validé avec succès')
    } catch (err) {
      error('Erreur lors de la validation')
    }
  }
}

D. FERMETURE AVEC MODIFICATIONS NON SAUVEGARDÉES
-------------------------------------------------
const handleClose = async () => {
  const hasChanges = isDirty()
  const canClose = await confirmExit(hasChanges)
  
  if (canClose) {
    closeModal()
  }
}

7. TYPES DE NOTIFICATIONS
═══════════════════════════════════════════════════════════════════════════════

SUCCESS (Vert)
--------------
- Création réussie
- Modification enregistrée
- Suppression effectuée
- Export terminé
- Validation réussie
- Envoi réussi

ERROR (Rouge)
-------------
- Échec de validation
- Erreur serveur
- Champs manquants
- Format invalide
- Opération échouée
- Connexion perdue

WARNING (Orange)
----------------
- Données non sauvegardées
- Limite atteinte
- Action importante
- Confirmation requise
- Données incomplètes

INFO (Bleu)
-----------
- Chargement en cours
- Information générale
- Aide contextuelle
- État du système
